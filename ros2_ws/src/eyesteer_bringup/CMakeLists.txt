cmake_minimum_required(VERSION 3.8)
project(eyesteer_bringup)
find_package(ament_cmake REQUIRED)
# record the repository root so installed nodes can import eyesteer_core (see */_root.py)
get_filename_component(EYESTEER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." REALPATH)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/eyesteer_root.txt" "${EYESTEER_ROOT}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/eyesteer_root.txt" DESTINATION share/${PROJECT_NAME})
install(DIRECTORY launch params config maps behavior_trees rviz DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(launch_testing_ament_cmake QUIET)
endif()
ament_package()
